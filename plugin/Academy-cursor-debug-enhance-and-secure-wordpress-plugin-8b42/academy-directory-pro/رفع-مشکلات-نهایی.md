#### ج) پیاده‌سازی عملکرد JavaScript

**فایل:** `assets/js/views.js`

```javascript
// دکمه Wishlist (علاقه‌مندی‌ها)
$(document).on('click', '.adp-btn-wishlist', function(e) {
    e.preventDefault();
    var $btn = $(this);
    var academyId = $btn.data('id');
    
    $btn.toggleClass('active');
    
    if ($btn.hasClass('active')) {
        $btn.find('i').removeClass('fa-regular').addClass('fa-solid');
        self.showToast('✅ به علاقه‌مندی‌ها اضافه شد');
    } else {
        $btn.find('i').removeClass('fa-solid').addClass('fa-regular');
        self.showToast('❌ از علاقه‌مندی‌ها حذف شد');
    }
    
    self.toggleWishlist(academyId);  // ذخیره در localStorage
});

// دکمه Compare (مقایسه)
$(document).on('click', '.adp-btn-compare', function(e) {
    e.preventDefault();
    var $btn = $(this);
    var academyId = $btn.data('id');
    
    $btn.toggleClass('active');
    
    if ($btn.hasClass('active')) {
        self.showToast('✅ به مقایسه اضافه شد');
    } else {
        self.showToast('❌ از مقایسه حذف شد');
    }
    
    self.toggleCompare(academyId);  // ذخیره در localStorage
});
```

#### د) ذخیره‌سازی با localStorage

```javascript
// Wishlist
toggleWishlist: function(academyId) {
    var wishlist = this.getWishlist();
    var index = wishlist.indexOf(academyId);
    
    if (index > -1) {
        wishlist.splice(index, 1);  // حذف
    } else {
        wishlist.push(academyId);   // اضافه
    }
    
    localStorage.setItem('adp_wishlist', JSON.stringify(wishlist));
}

// Compare (حداکثر 4 آیتم)
toggleCompare: function(academyId) {
    var compare = this.getCompare();
    var index = compare.indexOf(academyId);
    
    if (index > -1) {
        compare.splice(index, 1);
    } else {
        if (compare.length >= 4) {
            this.showToast('⚠️ حداکثر 4 آموزشگاه قابل مقایسه است');
            return;
        }
        compare.push(academyId);
    }
    
    localStorage.setItem('adp_compare', JSON.stringify(compare));
}
```
